    53.最大子数组和

    线性DP

    class Solution
{
public:
    int maxSubArray(vector<int> &nums)
    {
        int max_sum = nums[0];
        int pre_sum = nums[0];
        int n = nums.size();
        for (int i = 1; i < n; i++)
        {
            pre_sum = nums[i] + max(pre_sum, 0);
            max_sum = max(max_sum, pre_sum);
        }
        return max_sum;
    }
};
